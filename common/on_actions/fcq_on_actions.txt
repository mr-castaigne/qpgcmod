on_actions = {
    on_ruling_party_change = {
        effect = {
            if = {
                limit = {
                    original_tag = FCQ
                    has_government = dystopianism
                }
                set_cosmetic_tag = FCQ_dystopianism
            }
            else_if = {
                limit = {
                    original_tag = FCQ
                    has_government = reactionary
                }
                set_cosmetic_tag = FCQ_reactionary
            }
            else = {
                drop_cosmetic_tag = yes
            }
        }
    }
}
